    <script>
        const apiUrl = window.location.origin + "/qkLANfile/api/screenshot";
        
        const btnSingles = document.querySelectorAll('.btn-single');
        const btnPolls = document.querySelectorAll('.btn-poll');
        const btnCtrls = document.querySelectorAll('.btn-ctrl');
        const btnWrites = document.querySelectorAll('.btn-write');
        const btnCopies = document.querySelectorAll('.btn-copy');
        const btnHtmls = document.querySelectorAll('.btn-html');

        const imgEl = document.getElementById('screenshot');
        const placeholderEl = document.getElementById('placeholder');
        const ctrlPanelEl = document.getElementById('ctrl-panel');
        const imageWrapperEl = document.getElementById('image-wrapper');

        let mouseState = { Left: false, Middle: false, Right: false };
        let lastOpTime = Date.now();
        let lastDragTime = 0;
        let lastSendX = -1, lastSendY = -1;

        let isPolling = false;
        let pollController = null;
        let screenshotCoordRange = null;
        let ctrlState = 0; // 0:关闭, 1: 1倍, 2: 1.5倍
        let currentHtmlUrl = ""; // 保存API返回的URL

        // ==========================================
        // 核心增强函数
        // ==========================================
        
        // 1. Toast 提示
        function showToast(msg, type = 'info') {
            const container = document.getElementById('toast-container');
            const toast = document.createElement('div');
            toast.className = 'toast';
            toast.style.backgroundColor = type === 'error' ? 'rgba(231,76,60,0.9)' : (type === 'warning' ? 'rgba(243,156,18,0.9)' : 'rgba(0,0,0,0.75)');
            toast.textContent = msg;
            container.appendChild(toast);
            
            setTimeout(() => {
                toast.style.opacity = '0';
                setTimeout(() => toast.remove(), 300);
            }, 2500);
        }

        // 2. 双通道状态更新
        function updateStatus(msg, isError = false, type = 'capture') {
            const color = isError ? "#e74c3c" : "#555";
            const text = type === 'capture' ? `截图: ${msg}` : `操控: ${msg}`;
            
            const elDesktop = document.getElementById(type === 'capture' ? 'status-capture-desktop' : 'status-ctrl-desktop');
            const elMobile = document.getElementById(type === 'capture' ? 'status-capture-mobile' : 'status-ctrl-mobile');
            
            if (elDesktop) { elDesktop.textContent = text; elDesktop.style.color = color; }
            if (elMobile) { elMobile.textContent = text; elMobile.style.color = color; }
        }

        function sendMouseCmd(cmd) {
            lastOpTime = Date.now();
            fetch(window.location.origin + "/qkLANfile/mouse", {
                method: 'POST',
                body: cmd
            }).catch(e => showToast("指令发送失败: " + e.message, "error"));
        }

        async function fetchImage(signal = null) {
            const url = `${apiUrl}?t=${Date.now()}`;
            try {
                const response = await fetch(url, { signal: signal });
                if (response.status === 200) {
                    const blob = await response.blob();
                    if (imgEl.src) URL.revokeObjectURL(imgEl.src);
                    
                    imgEl.src = URL.createObjectURL(blob);
                    imgEl.style.display = 'block';
                    placeholderEl.style.display = 'none';
                    
                    let fileName = response.headers.get("X-File-Name");
                    try {
                        fileName = fileName ? decodeURIComponent(fileName) : "screenshot.jpg";
                    } catch (e) {
                        fileName = "screenshot.jpg";
                    }
                    
                    if (!screenshotCoordRange && fileName) {
                        const underscoreIdx = fileName.indexOf('_');
                        const dotIdx = fileName.lastIndexOf('.');
                        if (underscoreIdx !== -1 && dotIdx !== -1 && dotIdx > underscoreIdx) {
                            const rangeStr = fileName.substring(underscoreIdx + 1, dotIdx);
                            const parts = rangeStr.split('-').map(Number);
                            if (parts.length === 4 && parts.every(n => !isNaN(n))) {
                                screenshotCoordRange = { x1: parts[0], y1: parts[1], x2: parts[2], y2: parts[3] };
                            }
                        }
                    }
                    
                    updateStatus(`已成功获取 [${fileName}] - ${new Date().toLocaleTimeString()}`, false, 'capture');
                    return true;
                } else if (response.status === 204) {
                    updateStatus("等待中 (204 No Content)...", false, 'capture');
                    return false;
                } else if (response.status === 403) {
                    updateStatus("请求被拒绝 (403 Forbidden)！", true, 'capture');
                    return false;
                } else {
                    updateStatus(`请求失败: ${response.status}`, true, 'capture');
                    return false;
                }
            } catch (err) {
                if (err.name === 'AbortError') {
                    updateStatus("已手动中止。", false, 'capture');
                } else {
                    updateStatus(`网络异常: ${err.message}`, true, 'capture');
                }
                throw err;
            }
        }

        btnSingles.forEach(btn => {
            btn.addEventListener('click', async () => {
                if (isPolling) {
                    showToast("长轮询正在运行中！", "warning");
                    return;
                }
                btnSingles.forEach(b => b.disabled = true);
                updateStatus(`正在发送单次请求...`, false, 'capture');
                try {
                    // 首次尝试获取
                    let success = await fetchImage();
                    
                    // 如果返回 204 (即 success 为 false)，说明后端正在生成截图，自动重试最多 3 次
                    let retries = 0;
                    while (!success && retries < 3) {
                        retries++;
                        updateStatus(`截图生成中，正在进行第 ${retries} 次重试...`, false, 'capture');
                        await new Promise(resolve => setTimeout(resolve, 600)); // 等待 0.6 秒后重试
                        success = await fetchImage();
                    }
                    
                    if (!success) {
                        showToast("未检测到新截图，请稍后再试", "warning");
                    }
                } finally {
                    btnSingles.forEach(b => b.disabled = false);
                }
            });
        });

        btnPolls.forEach(btn => {
            btn.addEventListener('click', async () => {
                if (isPolling) {
                    isPolling = false;
                    btnPolls.forEach(b => b.classList.remove('polling'));
                    btnSingles.forEach(b => b.disabled = false);
                    if (pollController) pollController.abort();
                    showToast("已停止长轮询");
                } else {
                    isPolling = true;
                    btnPolls.forEach(b => b.classList.add('polling'));
                    btnSingles.forEach(b => b.disabled = true);
                    updateStatus(`长轮询已启动...`, false, 'capture');
                    showToast("长轮询开启");

                    while (isPolling) {
                        pollController = new AbortController();
                        try {
                            await fetchImage(pollController.signal);
                            if (isPolling && ctrlPanelEl.style.display !== 'none') {

                            }

                                break;

                                await new Promise(resolve => setTimeout(resolve, 1000));

                        }
                    }
                }

        function getDisplayedImageRect() {
           if (imgEl.style.display === 'none' || !imgEl.src) return imageWrapperEl.getBoundingClientRect();
            
            const rect = imgEl.getBoundingClientRect();
            const nw = imgEl.naturalWidth;
            const nh = imgEl.naturalHeight;
            if (!nw || !nh) return rect;

            const isRotated = window.matchMedia("(orientation: portrait)").matches;
            const imgRatio = nw / nh;
            const tagW = imgEl.offsetWidth;
            const tagH = imgEl.offsetHeight;
            const boxRatio = tagW / tagH;
            
            let renderW, renderH;
            if (imgRatio > boxRatio) {
                renderW = tagW; renderH = tagW / imgRatio;
            } else {
                renderH = tagH; renderW = tagH * imgRatio;
            }
            
            const screenW = isRotated ? renderH : renderW;
            const screenH = isRotated ? renderW : renderH;
            const left = rect.left + (rect.width - screenW) / 2;
            const top = rect.top + (rect.height - screenH) / 2;
            
            return { left: left, top: top, width: screenW, height: screenH, right: left + screenW, bottom: top + screenH };
        }

        function movePanelTo(targetCenterX, targetCenterY, actionName = "操控中...") {
            if (ctrlPanelEl.style.display === 'none') return;

            const bounds = getDisplayedImageRect();
            const panelHalfW = ctrlPanelEl.offsetWidth / 2;
            const panelHalfH = ctrlPanelEl.offsetHeight / 2;

            let newCenterX = Math.max(bounds.left, Math.min(targetCenterX, bounds.right));
            let newCenterY = Math.max(bounds.top, Math.min(targetCenterY, bounds.bottom));

            let newLeft = newCenterX - panelHalfW;
            let newTop = newCenterY - panelHalfH;

            ctrlPanelEl.style.left = newLeft + 'px';
            ctrlPanelEl.style.top = newTop + 'px';

            let screenX = Math.round(newCenterX);
            let screenY = Math.round(newCenterY);

            if (screenshotCoordRange && bounds.width > 0 && bounds.height > 0) {
                const relX = newCenterX - bounds.left;
                const relY = newCenterY - bounds.top;
                
                const isRotated = window.matchMedia("(orientation: portrait)").matches;
                let ratioX, ratioY;

                if (isRotated) {
                    ratioX = Math.max(0, Math.min(1, relY / bounds.height));
                    ratioY = Math.max(0, Math.min(1, 1 - (relX / bounds.width)));
                } else {
                    ratioX = Math.max(0, Math.min(1, relX / bounds.width));
                    ratioY = Math.max(0, Math.min(1, relY / bounds.height));
                }
                
                screenX = Math.round(screenshotCoordRange.x1 + ratioX * (screenshotCoordRange.x2 - screenshotCoordRange.x1));
                screenY = Math.round(screenshotCoordRange.y1 + ratioY * (screenshotCoordRange.y2 - screenshotCoordRange.y1));
            }

            updateStatus(`${actionName} 屏幕坐标: (${screenX}, ${screenY})`, false, 'ctrl');

            sendMouseCmd(`MVP:${screenX},${screenY}`);
            lastSendX = screenX;
            lastSendY = screenY;
            lastOpTime = Date.now();
        }

        function bindButtonHandler(btnEl, onClick, onLongPress = null) {
            let timer = null, interval = null, isLongPress = false;
            const clearTimers = () => { if (timer) clearTimeout(timer); if (interval) clearInterval(interval); timer = null; interval = null; };

            btnEl.addEventListener('pointerdown', (e) => {
                e.stopPropagation(); isLongPress = false; lastOpTime = Date.now();
                if (onLongPress) {
                    timer = setTimeout(() => { isLongPress = true; onLongPress(); interval = setInterval(() => { onLongPress(); }, 500); }, 400);
                }
            });
            btnEl.addEventListener('pointerup', (e) => { e.stopPropagation(); lastOpTime = Date.now(); if (!isLongPress) onClick(); clearTimers(); });
            btnEl.addEventListener('pointercancel', clearTimers);
            btnEl.addEventListener('pointerleave', clearTimers);
        }

        function toggleMouseKey(key, btnEl) {
            if (mouseState[key]) {
                sendMouseCmd(`MU:${key}`); mouseState[key] = false; btnEl.classList.remove('jumping');
            } else {
                sendMouseCmd(`MD:${key}`); mouseState[key] = true; btnEl.classList.add('jumping');
            }
        }

        function triggerMouseClick(key) {
            if (mouseState[key]) {
                sendMouseCmd(`MU:${key}\nDL:50\nMC:${key}`); mouseState[key] = false;
                const topKeyMap = { Left: 'btn-top-1', Middle: 'btn-top-3', Right: 'btn-top-5' };
                const topBtn = document.getElementById(topKeyMap[key]);
                if (topBtn) topBtn.classList.remove('jumping');
            } else { sendMouseCmd(`MC:${key}`); }
        }

        bindButtonHandler(document.getElementById('btn-top-1'), () => toggleMouseKey('Left', document.getElementById('btn-top-1')));
        bindButtonHandler(document.getElementById('btn-top-2'), () => sendMouseCmd('MW:30'), () => sendMouseCmd('MW:60'));
        bindButtonHandler(document.getElementById('btn-top-3'), () => toggleMouseKey('Middle', document.getElementById('btn-top-3')));
        bindButtonHandler(document.getElementById('btn-top-4'), () => sendMouseCmd('MW:-30'), () => sendMouseCmd('MW:-60'));
        bindButtonHandler(document.getElementById('btn-top-5'), () => toggleMouseKey('Right', document.getElementById('btn-top-5')));

                // 点击后立即将桌面端和移动端的C按钮置灰禁用

                showToast("正在获取选中文本...", "info");

                    const res = await fetch(window.location.origin + "/qkLANfile/clip/getselhtml");
                    if (!res.ok) throw new Error(`HTTP ${res.status}`);
                    const data = await res.json();
                    
                    if (data.text) {
                        if (navigator.clipboard && window.isSecureContext) {
                            await navigator.clipboard.writeText(data.text);
                        } else {
                            let textArea = document.createElement("textarea");
                            textArea.value = data.text;
                            document.body.appendChild(textArea);
                            textArea.select();
                            document.execCommand("copy");
                            document.body.removeChild(textArea);
                        }
                        showToast("选中文本已复制到剪贴板！"+data.text);

                        btnHtmls.forEach(b => b.style.display = 'flex');

                        btnHtmls.forEach(b => b.style.display = 'none');
